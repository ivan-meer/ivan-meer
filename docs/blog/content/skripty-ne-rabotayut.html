Сценарий, знакомый каждому, кто хоть раз внедрял AI-автоматизацию: вы подключаете LLM к API, пишете 200 строк на Python, демонстрируете заказчику — и система работает… ровно до первого нестандартного запроса.

Через неделю агент «забывает» контекст, через месяц счёт за токены вырастает втрое, а через три — код лежит мёртвым грузом, потому что его страшно трогать.

Это не невезение. Это отсутствие инженерного подхода.

Чем скрипт отличается от системы

Скрипт — это линейная последовательность действий. Он отвечает на запрос, пока запрос укладывается в ожидания автора. Система — это набор слоёв, каждый из которых решает свою задачу и умеет обрабатывать сбои.

Харнесс — это «операционная система» для AI-агента. Он управляет жизненным циклом запроса, от приёма до ответа, проходя через пять слоёв архитектуры.

Пример на Python: от скрипта к системе с оркестрацией

Посмотрим, как превратить скрипт в настоящую систему. Начнём с проблемы — тот самый «простой агент» из демо:

Теперь — то же самое, но как архитектура с харнессом:

# ✅ Система: харнесс с оркестрацией
from enum import Enum
from dataclasses import dataclass, field
from typing import Optional
import asyncio
import logging
import json

logger = logging.getLogger("agent.harness")

class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    RETRYING = "retrying"
    COMPLETED = "completed"
    FAILED = "failed"
    FALLBACK = "fallback"

@dataclass
class Task:
    id: str
    payload: dict
    status: TaskStatus = TaskStatus.PENDING
    retry_count: int = 0
    max_retries: int = 3
    result: Optional[str] = None
    error: Optional[str] = None
    metadata: dict = field(default_factory=dict)

class RetryHandler:
    """Обработчик retry-политик с exponential backoff."""

    def __init__(self, base_delay: float = 1.0, max_delay: float = 60.0):
        self.base_delay = base_delay
        self.max_delay = max_delay

    async def execute_with_retry(
        self, task: Task, action, fallback=None
    ) -> Task:
        while task.status in {TaskStatus.PENDING, TaskStatus.RETRYING}:
            try:
                task.status = TaskStatus.RUNNING
                task.result = await action(task)
                task.status = TaskStatus.COMPLETED
                return task

            except Exception as e:
                task.retry_count += 1
                task.error = str(e)
                logger.warning(
                    f"Task {task.id} failed (attempt {task.retry_count}): {e}"
                )

                if task.retry_count >= task.max_retries:
                    if fallback:
                        task.status = TaskStatus.FALLBACK
                        task.result = await fallback(task)
                        task.status = TaskStatus.COMPLETED
                        return task
                    task.status = TaskStatus.FAILED
                    return task

                task.status = TaskStatus.RETRYING
                delay = min(
                    self.base_delay * (2 ** (task.retry_count - 1)),
                    self.max_delay
                )
                await asyncio.sleep(delay)

        return task

class TaskQueue:
    """Очередь задач с приоритетами."""

    def __init__(self, max_concurrent: int = 5):
        self.queue: asyncio.PriorityQueue = asyncio.PriorityQueue()
        self.max_concurrent = max_concurrent
        self.active_tasks: set = set()

    async def enqueue(self, task: Task, priority: int = 0):
        await self.queue.put((priority, task))
        logger.info(f"Task {task.id} enqueued (priority={priority})")

    async def worker(self, processor):
        while True:
            _, task = await self.queue.get()
            self.active_tasks.add(task.id)
            try:
                await processor(task)
            finally:
                self.active_tasks.discard(task.id)
                self.queue.task_done()

    async def process_all(self, processor):
        workers = [
            asyncio.create_task(self.worker(processor))
            for _ in range(self.max_concurrent)
        ]
        await self.queue.join()
        for w in workers:
            w.cancel()

class AgentHarness:
    """Харнесс AI-агента — объединяет оркестрацию, retry и наблюдаемость."""

    def __init__(self):
        self.queue = TaskQueue(max_concurrent=5)
        self.retry = RetryHandler(base_delay=1.0, max_delay=30.0)
        self.metrics = MetricsCollector()

    async def process_task(self, task: Task):
        """Обработка задачи с полным циклом: retry, fallback, метрики."""
        start = asyncio.get_event_loop().time()

        await self.retry.execute_with_retry(
            task,
            action=self._execute_task,
            fallback=self._fallback_response,
        )

        duration = asyncio.get_event_loop().time() - start
        self.metrics.record(
            task_id=task.id,
            status=task.status.value,
            duration_ms=int(duration * 1000),
            retries=task.retry_count,
        )
        logger.info(
            f"Task {task.id} completed: {task.status.value} "
            f"({duration:.2f}s, {task.retry_count} retries)"
        )

    async def _execute_task(self, task: Task) -> str:
        """Основная логика выполнения задачи."""
        prompt = f"Ты помощник. Ответь: {task.payload['query']}"

        response = await llm.invoke(prompt)
        task.metadata["tokens"] = estimate_tokens(response)

        if task.payload.get("parse_json") and "{" in response:
            try:
                json.loads(response)
            except ValueError:
                raise ValueError("Invalid JSON in response")

        return response

    async def _fallback_response(self, task: Task) -> str:
        """Fallback при исчерпании retry."""
        logger.warning(f"Using fallback for task {task.id}")
        return (
            "Извините, я временно не могу обработать запрос. "
            "Пожалуйста, попробуйте позже."
        )

    async def run(self, query: str, parse_json: bool = False) -> str:
        """Запускает задачу через харнесс."""
        task = Task(
            id=generate_id(),
            payload={"query": query, "parse_json": parse_json},
        )
        await self.queue.enqueue(task, priority=0)
        await self.process_task(task)
        return task.result or task.error or "Unknown error"

    def get_metrics(self) -> dict:
        return self.metrics.summary()

Retry-политики: exponential backoff и circuit breaker

Retry — не просто «повтори попытку». Это целая наука. Две основные стратегии:

Exponential Backoff

Каждая следующая попытка ждёт вдвое дольше предыдущей. Базовая задержка 1с, максимальная — 60с. Попытки: 1с → 2с → 4с → 8с → 16с → 32с → 60с (cap).

class ExponentialBackoff:
    """Exponential backoff с jitter для предотвращения thundering herd."""

    def __init__(self, base: float = 1.0, max_delay: float = 60.0):
        self.base = base
        self.max_delay = max_delay

    def get_delay(self, attempt: int) -> float:
        import random
        delay = min(self.base * (2 ** attempt), self.max_delay)
        jitter = random.uniform(0, delay * 0.1)
        return delay + jitter

    async def wait(self, attempt: int):
        delay = self.get_delay(attempt)
        await asyncio.sleep(delay)
        return delay

Circuit Breaker

Если сервис падает слишком часто — перестаём его дёргать и возвращаем fallback. После тайм-аута пробуем снова.

class CircuitBreaker:
    """Circuit breaker: защита от каскадных сбоев."""

    def __init__(self, threshold: int = 5, reset_timeout: float = 30.0):
        self.threshold = threshold
        self.reset_timeout = reset_timeout
        self.failures = 0
        self.state = "closed"  # closed / open / half-open
        self.last_failure_time = 0.0

    async def call(self, action, fallback=None):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.reset_timeout:
                self.state = "half-open"
            else:
                return await fallback() if fallback else None

        try:
            result = await action()
            self.failures = 0
            self.state = "closed"
            return result
        except Exception as e:
            self.failures += 1
            self.last_failure_time = time.time()
            if self.failures >= self.threshold:
                self.state = "open"
                logger.error(
                    f"Circuit breaker opened after {self.failures} failures"
                )
            raise
Когда использовать circuit breaker

Circuit breaker имеет смысл при интеграции с внешними API, БД, LLM-провайдерами. Если сервис упал, нет смысла долбить его retry'ми — это только усугубит проблему. Лучше вернуть fallback и попробовать позже.

# In-memory очередь для прототипа (asyncio.Queue)
class InMemoryTaskQueue:
    """Простая очередь для разработки и тестов."""

    def __init__(self):
        self.queue = asyncio.PriorityQueue()
        self.processed = []

    async def add(self, task: Task, priority: int = 0):
        await self.queue.put((priority, task))

    async def get(self) -> Task:
        _, task = await self.queue.get()
        return task

# Redis-очередь для продакшена
class RedisTaskQueue:
    """Очередь через Redis — персистентность и масштабирование."""

    def __init__(self, redis_client, queue_name: str = "agent_tasks"):
        self.redis = redis_client
        self.queue_name = queue_name

    async def add(self, task: Task, priority: int = 0):
        await self.redis.zadd(
            self.queue_name,
            {task.id: priority},
        )
        await self.redis.hset(
            f"{self.queue_name}:data",
            task.id,
            json.dumps(asdict(task)),
        )

    async def get(self) -> Optional[Task]:
        # Получаем задачу с наивысшим приоритетом (наименьшее число)
        result = await self.redis.zpopmin(self.queue_name, count=1)
        if not result:
            return None
        task_id, _ = result[0]
        data = await self.redis.hget(f"{self.queue_name}:data", task_id)
        return Task(**json.loads(data)) if data else None

Главный вывод

AI-агент — это не «LLM + API». Это система с памятью, оркестрацией и наблюдаемостью.

Без этих трёх слоёв вы получаете не агента, а дорогой скрипт, который ломается при первой нестандартной ситуации. С ними — инженерный продукт, который можно передать команде, масштабировать и не бояться трогать.

Минимальный чеклист для продакшен-агента

Прежде чем выпускать агента в продакшен, проверьте:

  1. Есть ли retry-политика с exponential backoff?
  2. Есть ли circuit breaker для внешних сервисов?
  3. Логируется ли каждый шаг с контекстом?
  4. Собираются ли метрики latency, error rate, token usage?
  5. Есть ли fallback при недоступности LLM?
  6. Работает ли очередь задач (не блокирующая)?
  7. Можно ли перезапустить систему без потери контекста?
